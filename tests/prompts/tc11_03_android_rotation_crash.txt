void Renderer::setWindow(ANativeWindow* newWindow) {
    if (window) ANativeWindow_release(window);
    window = newWindow;
    if (window) createSurfaceAndSwapchain();
}

void Renderer::drawFrame() {
    // render thread keeps running even if window == nullptr
    vkAcquireNextImageKHR(device, swapchain, UINT64_MAX, imageAvailable, VK_NULL_HANDLE, &imageIndex);
    vkQueueSubmit(queue, 1, &submit, frameFence);
    vkQueuePresentKHR(queue, &present);
}

void Renderer::createSurfaceAndSwapchain() {
    vkCreateAndroidSurfaceKHR(instance, &surfaceInfo, nullptr, &surface);
    vkCreateSwapchainKHR(device, &swapInfo, nullptr, &swapchain);
    createSwapchainImageViews();
    createDepthImages();
    createFramebuffers();
}
